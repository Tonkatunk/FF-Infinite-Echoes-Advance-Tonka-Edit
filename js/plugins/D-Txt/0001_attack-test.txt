//敵だったら点滅表示させる
<D-Animation:self>
condition = a.isEnemy();
id = 22;
</D-Animation>

//--ミス・ブロッキング--------------------------------------------------------------------------------------------

// 攻撃が命中しなかった場合、ミスの効果音を鳴らす
<D-Motion>
condition = b._results[0].missed;
playSe = {"name":"FF7 cancel","volume":90,"pitch":100,"pan":0}
</D-Motion>

// 武器1ブロッキング
<D-Motion:target&svBlockR>
condition = b.isActor() && b.isWeapon1Block(b._results[0].evaRand);
playSe = {"name":"FF6 guard2","volume":90,"pitch":100,"pan":0}
</D-Motion>

// 武器2or盾ブロッキング
<D-Motion:target&svBlockL>
condition = b.isActor() && ( b.isWeapon2Block(b._results[0].evaRand) || b.isSealdBlock(b._result.evaRand));
playSe = {"name":"FF6 guard2","volume":90,"pitch":100,"pan":0}
</D-Motion>

// アクセサリーブロッキング
<D-Motion:target&svMant>
condition = b.isActor() && b.isAccessoryBlock(b._results[0].evaRand);
playSe = {"name":"MZ Equip3","volume":90,"pitch":100,"pan":0}
</D-Motion>

// マテリアブロッキング
<D-Motion:target&svSpell>
condition = b.isActor() && b.isMateriaBlock(b._results[0].evaRand);
playSe = {"name":"FF6 guard2","volume":90,"pitch":100,"pan":0}
</D-Motion>

// 通常回避モーション
<D-Motion:target&svMissile>
condition = b.isActor() && b.isNormalEva(b._results[0].isNormalEva);
playSe = {"name":"FF7 cancel","volume":90,"pitch":100,"pan":0}
</D-Motion>

// -- 前進しつつ攻撃モーション -----------

//nowaitを使うと何故かattackが作用しないのでつけない
// -- 振りかざす武器ならとびかかるモーションを再生 -----------
<D-Motion:attack>
condition = a.isActor() && a.attackMotionType() == "swing";
arcY = -$TILE/2;
ex = a.x-$TILE;
</D-Motion>

// -- 振りかざさない武器ならとびかからず前進のみ -----------
<D-Motion:attack>
condition = a.isActor() && a.attackMotionType(false) != "swing";
ex = a.x-$TILE;
</D-Motion>

// 投擲一投目
<D-Animation:noWait>
condition = a.attackAnimationType(false) == "throw";
dx = a.getBoomelanPoint(a.x+a.width/2,a.y-a.height/2,b.x,b.y-b.height/2,t,36,$TILE)[0];
dy = a.getBoomelanPoint(a.x+a.width/2,a.y-a.height/2,b.x,b.y-b.height/2,t,36,$TILE)[1];
id = a.attackAnimationId1(); // 一撃目のID
</D-Animation>

// 投擲一投目の斬撃アニメーション
<D-Animation:noWait>
condition = a.isActor() && !b._results[0].missed && !b._results[0].evade && a.attackAnimationType(false) == "throw";
id = 16
</D-Animation>

// 攻撃が成功した場合はアニメーションを流す(通常武器)
<D-Animation>
condition = !b._results[0].missed && !b._results[0].evaded && a.attackAnimationType(false) == "normal";
id = a.attackAnimationId1(); // 一撃目のID
wait = 18;
</D-Animation>

// 攻撃が成功した場合はアニメーションを流す(弓矢)
<D-Animation:noWait&shot>
condition = !b._results[0].missed && !b._results[0].evaded && a.attackAnimationType(false) == "bow";
arrival = 18;
sy = a.y - $DOT*6;
id = a.attackAnimationId1(); // 一撃目のID
</D-Animation>

// 二撃目 --------------------------------------------------------------------
// アクターかつ二撃目のアニメーションがある場合、攻撃モーションを再生
<D-Motion:delay&LAttack>
delay = -5
condition = a.isActor() && a.attackAnimationId2() && a.weapons()[1]
weaponType = a.weapons()[1].wtypeId // 二撃目の武器タイプ
</D-Motion>

// 攻撃が命中した(not回避されなかった)場合はアニメーションを流す
<D-Animation:noWait>
condition = a.isActor() && a.weapons()[1] && a.attackAnimationId2() && !b._results[1].missed && !b._results[1].evade && a.attackMotionType(false) == "swing" && a.attackAnimationType(true) != "throw";
id = a.attackAnimationId2() // 二撃目のID
</D-Animation>

// 投擲二投目
<D-Animation:noWait>
condition = a.attackAnimationType(true) == "throw";
dx = a.getBoomelanPoint(a.x+a.width/2,a.y-a.height/2,b.x,b.y-b.height/2,t,36,$TILE)[0];
dy = a.getBoomelanPoint(a.x+a.width/2,a.y-a.height/2,b.x,b.y-b.height/2,t,36,$TILE)[1];
id = a.attackAnimationId2(); // 二撃目のID
frame = 18 * (a.attackAnimationType(true) == "throw")/1
</D-Animation>

// 投擲二投目の斬撃アニメーション
<D-Animation:noWait>
condition = a.isActor() && a.weapons()[1] && a.attackAnimationId2() && !b._results[1].missed && !b._results[1].evade && a.attackAnimationType(true) == "throw";
id = 16
</D-Animation>

<D-Animation:noWait>
id = 1;
</D-Animation>
